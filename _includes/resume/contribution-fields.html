{% comment %} Sets res_title, res_desc, res_achievements and res_labels for include.contribution, in the page language. English reads the "en" columns of Notion, the print ones as fallback, and French when neither is set. Achievements come from the Items database: one without English name is left out rather than shown in French. Labels go through resume_page.tags in _data/translations.yml. {% endcomment %} {% assign res_lang = page.lang | default: site.default_lang | default: 'fr' %} {% assign contribution = include.contribution %} {% if res_lang == 'en' %} {% if contribution.title_en.size > 0 %} {% assign res_title = contribution.title_en %} {% elsif contribution.title_print_en.size > 0 %} {% assign res_title = contribution.title_print_en %} {% else %} {% assign res_title = contribution.title %} {% endif %} {% if contribution.description_en.size > 0 %} {% assign res_desc = contribution.description_en %} {% elsif contribution.description_print_en.size > 0 %} {% comment %} The print text starts lowercase, after a " · " {% endcomment %} {% assign res_first = contribution.description_print_en | slice: 0 | upcase %} {% assign res_rest = contribution.description_print_en | slice: 1, 10000 %} {% assign res_desc = res_first | append: res_rest %} {% else %} {% assign res_desc = contribution.description %} {% endif %} {% assign res_achievements = '' | split: '' %} {% for item_id in contribution.item_ids %} {% assign res_item = site.data.notion_items | where: 'id', item_id | first %} {% if res_item.name_en.size > 0 %} {% assign res_achievements = res_achievements | push: res_item.name_en %} {% endif %} {% endfor %} {% assign res_labels = '' | split: '' %} {% for label in contribution.labels %} {% assign res_label = site.data.translations.en.resume_page.tags[label] | default: label %} {% assign res_labels = res_labels | push: res_label %} {% endfor %} {% else %} {% assign res_title = contribution.title %} {% assign res_desc = contribution.description %} {% assign res_achievements = contribution.achievements %} {% assign res_labels = contribution.labels %} {% endif %}
